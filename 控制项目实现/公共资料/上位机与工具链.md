## 1. 为什么数据链比算法重要

同样一个 PID，两种人：

| | A | B |
|---|---|---|
| 调参依据 | 「看着好像稳了」 | 阶跃曲线 + 超调 8.2% + 上升 76 ms |
| 出问题 | 重烧固件试一遍 | 回看 CSV，发现是周期抖了 2 ms |
| 产出 | 一个能动的车 | 一份可对比的实验记录 |

**B 才能在两周内调完 P3。** 上位机不是"锦上添花"，它决定你能不能在有限时间里收敛。

三条原则：

1. **采样数据要带时间戳**，不能靠上位机收包时间（串口有缓冲，会抖）。
2. **参数随数据一起存**，否则三个月后看不懂 `p3_test3_final.xlsx`。
3. **实时只用于"看趋势"，结论一律从落盘 CSV 算**。

## 2. 串口协议：两档方案

### 2.1 方案 A：CSV 文本（先从这个开始）

```
# MCU 每 100 ms 发一行
1713421234,300.0,298.4,8120,30.0,15.0,0.0\r\n
   时间戳ms  目标   实测  PWM   kp   ki  kd
```

- 优点：**人眼可读、Excel 能开、不用写解析器**，调试期效率最高。
- 缺点：约 30–40 字节/行，115200 波特率下最高约 250 行/s（够 10–20 Hz 上报）。
- 建议：**所有项目起步都用它**。P1 的 10 Hz、P5 的 1 Hz 都用这个就够。

```python
# host_csv.py
import serial, csv, time
ser = serial.Serial("COM5", 115200, timeout=0.1)
with open("run.csv", "w", newline="") as f:
    w = csv.writer(f); w.writerow(["t_ms","ref","meas","out","kp","ki","kd"])
    t0 = time.time()
    while time.time() - t0 < 20:                 # 录 20 秒
        line = ser.readline().decode(errors="ignore").strip()
        if line.count(",") == 6:
            w.writerow(line.split(","))
```

### 2.2 方案 B：二进制帧 + CRC（需要高速采样时）

200 Hz × 7 通道 = 1400 个数/s，文本扛不住，改二进制：

| 字节 | 内容 | 说明 |
|---|---|---|
| 0–1 | `0xAA 0x55` | 帧头（两个不同字节，避免误同步） |
| 2 | `len` | payload 字节数 |
| 3 | `type` | 0x01=遥测, 0x02=参数, 0x03=命令应答 |
| 4..4+len−1 | payload | float32 小端数组 |
| 末尾 2 | CRC16 | 覆盖 len/type/payload（CRC-16/CCITT） |

```c
/* MCU 侧：打包（不要用 printf） */
static uint8_t  tx_buf[128];
static uint16_t tx_len = 0;

void telemetry_push(const float *v, uint8_t n)
{
    uint8_t *p = tx_buf;
    *p++ = 0xAA; *p++ = 0x55;
    *p++ = (uint8_t)(n * 4);
    *p++ = 0x01;
    memcpy(p, v, n * 4); p += n * 4;
    uint16_t crc = crc16(tx_buf + 2, 2 + n * 4);
    *p++ = crc & 0xFF; *p++ = crc >> 8;
    HAL_UART_Transmit_DMA(&huart1, tx_buf, (uint16_t)(p - tx_buf));
}
```

```python
# host_bin.py 解析
import struct, serial
ser = serial.Serial("COM5", 921600, timeout=0.05)

def read_frame(ser):
    while True:
        if ser.read(1) != b"\xaa":      continue
        if ser.read(1) != b"\x55":      continue
        ln, typ = ser.read(1)[0], ser.read(1)[0]
        body = ser.read(ln)
        crc  = ser.read(2)
        if crc16(b"\x00" + bytes([ln, typ]) + body) != int.from_bytes(crc, "little"):
            continue                     # CRC 错，丢帧等下一个帧头
        return typ, struct.unpack(f"<{ln//4}f", body)
```

> **为什么必须有 CRC**：串口在电机噪声下会丢字节。没有校验时，一次错位会让 CSV 的所有列永久错位，你还以为"算法出问题了"。

## 3. MCU 侧的三个纪律

| 纪律 | 做法 | 违反的后果 |
|---|---|---|
| 控制环里不发串口 | 只在主循环发；或用 DMA + 环形缓冲 | 周期抖动 2–5 ms，积分/微分全错 |
| 不用阻塞 `printf` 重定向 | 用 `HAL_UART_Transmit_DMA`，或自己写 `ring_put` | 同上 |
| 采样用定时器 | TIM 中断触发 ADC/I2C，主循环只处理数据 | `dt` 不固定，`Q` 矩阵假设失效 |

**最省事的折中**（学生项目推荐）：控制环只把数据写进全局数组，主循环每 10 ms 检查并发送一次：

```c
volatile struct { float v[8]; uint8_t ready; } telem;

/* 1 ms 控制中断里: */
memcpy((void*)telem.v, snapshot, sizeof(snapshot));
telem.ready = 1;

/* 主循环: */
if (telem.ready) { telem.ready = 0; telemetry_push((float*)telem.v, 8); }
```

## 4. Python 上位机（实时看趋势）

用 **pyqtgraph**（比 matplotlib 实时快一个数量级）：

```python
# host_live.py —— 实时波形
import serial, numpy as np, pyqtgraph as pg
from pyqtgraph.Qt import QtCore, QtWidgets

ser = serial.Serial("COM5", 115200, timeout=0)
win = pg.GraphicsLayoutWidget(title="Realtime")
plots = []
for i, name in enumerate(["ref", "meas", "out"]):
    p = win.addPlot(row=i, col=0, title=name)
    p.setYRange(-400, 400) if i < 2 else p.setYRange(-10500, 10500)
    plots.append(p.plot(pen=pg.mkPen(width=2)))

buf = np.zeros((3, 1000))
idx = 0

def update():
    global idx
    while ser.in_waiting:
        line = ser.readline().decode(errors="ignore").strip()
        if line.count(",") != 6: continue
        _, ref, meas, out, *_ = line.split(",")
        buf[0, idx % 1000] = float(ref)
        buf[1, idx % 1000] = float(meas)
        buf[2, idx % 1000] = float(out)
        idx += 1
    for i in range(3):
        plots[i].setData(buf[i])

timer = QtCore.QTimer(); timer.timeout.connect(update); timer.start(30)
win.show(); pg.exec()
```

**只用它看形状**（是抖、是振、是超调）。**结论一律从 CSV 算。**

## 5. 统一的指标计算（四个项目共用一份代码）

```python
# metrics.py —— 所有项目的验收指标都从这里算，保证口径一致
import numpy as np

def step_metrics(t, y, ref, settle_band=0.02):
    """t: s, y: 实测, ref: 目标值(取末值). 返回标准指标字典."""
    ref = float(ref) if np.isscalar(ref) else float(ref[-1])
    y0  = float(y[0])
    sp  = abs(ref - y0)
    sgn = np.sign(ref - y0) or 1.0

    yn = (y - y0) * sgn / sp                 # 归一化到 0..1
    i10 = np.argmax(yn >= 0.1); i90 = np.argmax(yn >= 0.9)
    t_rise = t[i90] - t[i10]

    over = (np.max(yn) - 1.0) * 100.0        # 超调 %

    band = settle_band
    out_of_band = np.where(np.abs(yn - 1.0) > band)[0]
    i_set = out_of_band[-1] + 1 if len(out_of_band) else 0
    t_settle = t[i_set] - t[0]

    ss_err = (float(y[-1]) - ref) / abs(ref) * 100.0
    iae = np.trapz(np.abs(ref - y), t)       # 积分绝对误差

    return dict(t_rise_ms=t_rise*1000, overshoot_pct=over,
                t_settle_ms=t_settle*1000, ss_err_pct=ss_err, iae=iae)

if __name__ == "__main__":
    import pandas as pd
    df = pd.read_csv("run.csv")
    print(step_metrics(df.t_ms.values/1000.0, df.meas.values, df.ref.values))
```

把这份文件放到 `控制项目实现/tools/` 下，**P1/P3/P5 全部引用它**——这样四个项目的指标定义完全一致，横向对比才成立。

## 6. 在线调参（省下大量烧录时间）

调 P3 的 `Kp/Kd/gamma` 时，烧一次固件 30 秒，一天要烧 100 次。加个命令协议：

```
PC → MCU:  "S kp_theta 12.5\n"      # 设置
           "G\n"                     # 读回全部参数
           "R\n"                     # 复位到默认值
MCU → PC:  "OK kp_theta=12.50\n"     # 确认（带实际生效值）
```

```c
/* 主循环解析 */
if (cmd[0] == 'S') {
    char key[32]; float val;
    if (sscanf(cmd, "S %31s %f", key, &val) == 2 && param_set(key, val))
        printf("OK %s=%.2f\r\n", key, val);
    else
        printf("ERR unknown\r\n");
}
```

**纪律**：`param_set` 只改 RAM 里的值，不做 float 除法/乘法重算；参数结构体加 `const` 默认值表，`R` 命令能一键回到已验证的版本。

## 7. 参数巡参（把"调参"变成实验）

手动试 10 组 `Kp` 太慢。让 MCU 支持「自动跑一组参数」：

```
PC 发: "SCAN kp 20 60 5\n"     # kp 从 20 到 60, 步长 5
MCU:   每组参数跑 5 s 阶跃, 采集, 上报, 再跑下一组
```

```python
# scan.py —— 自动巡参并生成对比表
import serial, numpy as np, pandas as pd
from metrics import step_metrics

results = []
for kp in np.arange(20, 61, 5):
    data = run_one_trial(kp)                 # 你的采集函数
    m = step_metrics(data["t"], data["y"], data["ref"])
    results.append(dict(kp=kp, **m))

df = pd.DataFrame(results).sort_values("iae")
print(df.to_markdown(index=False))
```

**这一步在报告里特别加分**：不是「我调到 45 最好」，而是「我扫了 Kp ∈ [20,60]，IAE 在 40 处最小，如下表」——这是**实验**，不是**玄学**。

## 8. MATLAB / Simulink（可选，你有 `../../matlab实现/`）

两条可走的路：

| 路线 | 做法 | 适合 |
|---|---|---|
| **模型验证** | 把 A/B 矩阵导进 Simulink，和实测曲线叠图 | 快，验证建模对不对 |
| **嵌入式代码生成** | Simulink 画控制框图 → Embedded Coder 生成 C → 塞进 Keil 工程 | 想体验工业流程 |

**我的建议**：**只用 MATLAB 做「建模仿真 + 对比」，控制代码手写。** 理由是手写的代码你能调试、能进中断、能控制浮点开销；代码生成适合量产，不适合学习。

真正值得用 Simulink 的一件事：**Processor-in-the-Loop**——把生成的代码跑在真实板子上，用 Simulink 喂数据、看波形，省掉写上位机的功夫。有余力再玩。
